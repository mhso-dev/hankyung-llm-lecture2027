## 7. 부품 준비 — 고객 표를 데이터베이스에 넣기

def insert_customers(database_name: str, customers) -> list:

    rejected_ids = []

            rejected_ids.append(record["customer_id"])
    connection.commit()
    connection.close()
    return rejected_ids
